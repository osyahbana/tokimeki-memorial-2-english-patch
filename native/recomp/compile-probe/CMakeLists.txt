cmake_minimum_required(VERSION 3.20)
project(TM2CodegenCompileProbe C)
if(NOT IS_DIRECTORY "${FRAMEWORK}/runtime/include" OR NOT IS_DIRECTORY "${GENERATED}")
    message(FATAL_ERROR "Set FRAMEWORK and GENERATED to absolute existing directories")
endif()
file(GLOB SHARDS CONFIGURE_DEPENDS "${GENERATED}/*_full_*.c" "${GENERATED}/*_dispatch.c")
if(NOT SHARDS)
    message(FATAL_ERROR "No generated shards found")
endif()
add_library(tm2_codegen_probe OBJECT ${SHARDS})
target_include_directories(tm2_codegen_probe PRIVATE "${FRAMEWORK}/runtime/include" "${GENERATED}")
set_property(TARGET tm2_codegen_probe PROPERTY C_STANDARD 11)
